----DATABASE SCHEMA-----

CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100) UNIQUE,
    password_hash VARCHAR(255),
    role VARCHAR(20) CHECK (role IN ('manufacturer','distributor','pharmacy','customer','admin')),
    organization_name VARCHAR(100),
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE medicine_batches (
    id SERIAL PRIMARY KEY,
    batch_uid UUID DEFAULT gen_random_uuid() UNIQUE,
    drug_name VARCHAR(150),
    manufacturer_id INT REFERENCES users(id),
    quantity INT,
    manufacture_date DATE,
    expiry_date DATE,
    status VARCHAR(20) DEFAULT 'created',
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE custody_ledger (
    id SERIAL PRIMARY KEY,
    batch_id INT REFERENCES medicine_batches(id),
    event_type VARCHAR(30),
    actor_id INT REFERENCES users(id),
    actor_role VARCHAR(20),
    location VARCHAR(100),
    timestamp TIMESTAMP DEFAULT NOW(),
    previous_hash VARCHAR(64),
    current_hash VARCHAR(64)
);

CREATE TABLE transactions (
    id SERIAL PRIMARY KEY,
    batch_id INT REFERENCES medicine_batches(id),
    seller_id INT REFERENCES users(id),
    buyer_id INT REFERENCES users(id),
    quantity INT,
    transaction_date TIMESTAMP DEFAULT NOW()
);

CREATE TABLE flags (
    id SERIAL PRIMARY KEY,
    batch_id INT REFERENCES medicine_batches(id),
    reason VARCHAR(200),
    flagged_at TIMESTAMP DEFAULT NOW(),
    resolved BOOLEAN DEFAULT FALSE
);
